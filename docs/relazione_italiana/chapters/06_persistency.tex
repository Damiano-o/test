% CAPITOLO 6 — PERSISTENCY
\chapter{Persistenza}
\label{cap:persistenza}

La persistenza è intercambiabile a runtime (NFR-01) con tre backend tramite
il pattern \textbf{Abstract Factory}: \texttt{DAOFactory} +
concrete factory per modalità.

\section{Modalità}
\begin{itemize}
    \item \textbf{DEMO} (default): \texttt{DemoDAOFactory}, dati in-memory,
    seed idempotente. \texttt{DemoDAOTest}.
    \item \textbf{FS}: \texttt{FSDAOFactory}, entità su file JSON via
    \texttt{Gson}. \texttt{FSDaoTest}.
    \item \textbf{JDBC}: \texttt{JDBCDAOFactory}, database MySQL
    (\texttt{sql/CiboAmico.sql}). Escluso da JaCoCo (richiede MySQL), test
    manuale.
\end{itemize}

\section{Scelta a runtime}
\texttt{ApplicationModeManager} (Singleton) seleziona la factory secondo la
modalità attiva; i controller ottengono i DAO solo via
\texttt{getDAOFactory()}, mai con \texttt{new} diretto.

\begin{verbatim}
factory = switch (activeMode) {
    case MODE_JDBC -> new JDBCDAOFactory();
    case MODE_FS   -> new FSDAOFactory();
    case MODE_DEMO -> new DemoDAOFactory();
};
\end{verbatim}

\section{Connessione e transazioni (JDBC)}
\begin{itemize}
    \item \texttt{ConnectionManager}: credenziali da system properties
    (\texttt{ciboamico.db.url}, \texttt{ciboamico.db.user}, \texttt{ciboamico.db.password}) con fallback locale;
    \item transazioni atomiche con \texttt{setAutoCommit(false)} e
    \texttt{commit/rollback} (ordine + riduzione scorte).
\end{itemize}
